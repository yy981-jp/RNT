	// インデントの追加
	result.append(e.depth * 2, ' ');
	
	// 一時オブジェクトを作らず直接 result へ追加
	result.append("<\"");
	result.append(e.name);
	result.append("\">");
	if (e.isDir) result.push_back('/');
	
	result.append("          ");
	result.append(BaseNum::encode(e.id.value));
	result.push_back('\n');
